  const js=days.map(d=>S.journal[d]).filter(Boolean);
  if(js.length>=3&&sum(js,x=>x.energy)/js.length<=2.5&&ls.length>5)return'طاقتك واطية بقالك كام يوم وانت لسه بتحمّل جامد';
  const first=S.logs.length?S.logs[0].date:null;
  if(first&&diffDays(today(),first)>=42&&(!S.lastDeload||diffDays(today(),S.lastDeload)>=42))return'عدّى 6 أسابيع تدريب من غير أسبوع تخفيف';
  return null;
}
function analyze(){
  const T=targets(),days=[...Array(7)].map((_,i)=>addDays(today(),-i)),good=[],lack=[],tips=[],alerts=[];
  const fd=days.filter(d=>dayFoods(d).length);
  const avgK=fd.length?sum(fd,d=>tot(dayFoods(d)).k)/fd.length:0,avgP=fd.length?sum(fd,d=>tot(dayFoods(d)).p)/fd.length:0;
  const sl=days.map(d=>S.sleep[d]).filter(Boolean),avgS=sl.length?sum(sl,x=>x.h)/sl.length:0;
  const avgW=sum(days,waterMl)/7,wkDays=new Set(S.logs.filter(l=>days.includes(l.date)).map(l=>l.date)).size;
  const avgSt=sum(days,d=>S.steps[d]||0)/7,caf=days.map(d=>sum(S.caf.filter(x=>x.date===d),x=>x.mg));
  const goal=S.profile.goal;
  if(fd.length<4)lack.push(`مسجلتش أكل غير ${fd.length} أيام من 7 — كل ما تسجل أكتر Game Over يحللك أدق`);
  else{
    const r=avgK/T.kcal;
    if(goal==='bulk'){r>=.92?good.push(`سعراتك (${r0(avgK)}) قريبة من هدف التضخيم 👏`):lack.push(`بتاكل ${r0(avgK)} سعر والمطلوب ${r0(T.kcal)} — ضيف وجبة سناك فيها 300 سعر زي زبادي وموز ومكسرات`)}
    else if(goal==='cut'){r<=1.08?good.push('ملتزم بالعجز في السعرات 🔥'):lack.push(`سعراتك أعلى من هدف التنشيف بـ ${r0(avgK-T.kcal)} سعر يوميًا`)}
    else{Math.abs(r-1)<.1?good.push('سعراتك ثابتة على الصيانة'):lack.push('سعراتك بعيدة عن هدف الثبات')}
    avgP>=T.p*.9?good.push(`البروتين ممتاز (${r0(avgP)} جم)`):lack.push(`البروتين ${r0(avgP)} جم والمطلوب ${r0(T.p)} جم — زوّد بيض/فراخ/تونة أو سكوب واي`);
  }
  sl.length?(avgS>=7?good.push(`نومك كويس (${r1(avgS)} ساعة)`):lack.push(`نومك ${r1(avgS)} ساعة بس — النوم هو أهم عامل للتضخيم`)):lack.push('مسجلتش نوم الأسبوع ده');
  avgW>=T.water*.8?good.push('مية كفاية 💧'):lack.push(`المية متوسطها ${r1(avgW/1000)} لتر والمطلوب ${r1(T.water/1000)} لتر`);
  const planned=S.plan.length||S.profile.days;
  wkDays>=Math.max(1,planned*.8)?good.push(`التزمت بـ ${wkDays} تمرينات الأسبوع ده 💪`):lack.push(`تمرنت ${wkDays} أيام من ${planned} مخططة`);
  avgSt>=8000?good.push('حركتك اليومية ممتازة'):avgSt<5000&&lack.push('خطواتك قليلة — حاول توصل 7-8 آلاف خطوة');
  if(Math.max(...caf)>400){lack.push('الكافيين عدّى 400 ملجم في يوم — قلل عشان النوم');}
  const pl=plateau();if(pl){alerts.push(`⚖️ ${pl.msg}. ${goal==='bulk'?'زوّد 150-200 سعر في اليوم':goal==='cut'?'قلل 150 سعر أو زوّد كارديو':'لو هدفك التغيير عدّل السعرات'}.`)}
  const dl=deloadCheck();if(dl)alerts.push(`🛌 وقت أسبوع تخفيف (Deload): ${dl}. قلل السيتات للنص وخفف الأوزان 10%.`);
  const v=volume(),totalSets=sum(Object.values(v));
  if(totalSets>0){MUSCLES.forEach(m=>{const mn=m==='بطن'?4:8;if(v[m]<mn)alerts.push(`⚠️ ${m} مهمَل الأسبوع ده (${v[m]} سيت) — حاول توصل ${mn}-${mn+4}`);else if(v[m]>22)alerts.push(`🚨 ${m} حجمه عالي (${v[m]} سيت) — خد بالك من الإجهاد`)})}
  if(lack.length)tips.push('ابدأ بحاجة واحدة بس: '+lack[0].split(' — ')[0]);
  tips.push('كل أسبوع حاول تزوّد ريب أو وزن صغير في تمرين واحد على الأقل (Progressive Overload)');
  return{good,lack,tips,alerts};
}

/* ============ مولّد الخطط ============ */
const injured=ex=>S.injuries.some(z=>ex.avoid.includes(z));
const equipOk=ex=>!S.settings.home||ex.home||ex.eq==='body'||ex.eq==='band';
function pickEx(m,count,used){
  const pool=EXS.filter(e=>e.m===m&&!used.has(e.id)&&!injured(e)&&equipOk(e));
  if(pool.length<count)EXS.filter(e=>e.m===m&&!used.has(e.id)&&equipOk(e)).forEach(e=>{if(!pool.includes(e))pool.push(e)});
  const res=pool.slice(0,count);res.forEach(e=>used.add(e.id));return res;
}
function buildPlan(days,goal,style='hypertrophy'){
  const sets=goal==='bulk'?4:goal==='cut'?3:3;
  const reps=style==='strength'?'4-6':goal==='cut'?'10-15':'8-12';
  const rpe=goal==='cut'?8.5:8;
  const splits={
    3:[['صدر + تراي',['صدر','تراي']],[ 'ضهر + باي',['ضهر','باي']],['رجل + كتف + بطن',['رجل','كتف','بطن']]],
    4:[['علوي (Upper A)',['صدر','ضهر','كتف','باي']],['سفلي (Lower A)',['رجل','بطن']],['علوي (Upper B)',['كتف','صدر','ضهر','تراي']],['سفلي (Lower B)',['رجل','بطن']]],
    5:[['صدر',['صدر','تراي']],['ضهر',['ضهر','باي']],['رجل',['رجل']],['كتف + بطن',['كتف','بطن']],['دراعات + سمانة',['باي','تراي','رجل']]],
    6:[['دفع (Push A)',['صدر','كتف','تراي']],['سحب (Pull A)',['ضهر','باي']],['رجل (Legs A)',['رجل','بطن']],['دفع (Push B)',['كتف','صدر','تراي']],['سحب (Pull B)',['ضهر','باي','بطن']],['رجل (Legs B)',['رجل']]]
  };
  const sp=splits[days]||splits[4];
  const mapIdx=[0,1,2,3,4,5,6].map(i=>i<sp.length?i%sp.length:null);
  return mapIdx.map(idx=>{
    if(idx===null)return{name:'راحة واستشفاء',rest:true,ex:[]};
    const [name,ms]=sp[idx],used=new Set(),ex=[];
    const per=ms.length===2?3:ms.length===3?2:2;
    ms.forEach(m=>{pickEx(m,per,used).forEach(e=>ex.push({id:e.id,sets,reps,rpe}))});
    return{name,rest:false,ex};
  });
}
function buildDiet(T,mode='lean',days=4){
  const mult=mode==='dirty'?1.12:1,targetK=T.kcal*mult;
  const out=[];
  for(let i=0;i<days;i++){
    const meals=SLOTS.map(slot=>{
      const optsList=MO[slot];
      const opt=optsList[(i+SLOTS.indexOf(slot))%optsList.length];
      let baseK=0;
      const items=opt.items.map(([fn,g])=>{
        const f=findF(fn);if(!f)return null;
        const m=macrosOf(f,g);baseK+=m.k;
        return{name:f.n,g,sn:f.sn,sg:f.sg,...m};
      }).filter(Boolean);
      const scale=baseK>0?clamp((targetK*SLOTK[slot])/baseK,.5,2.2):1;
      const scaled=items.map(it=>{
        const g=r0(it.g*scale/5)*5||5;
        const f=findF(it.name);
        return{name:it.name,g,sn:it.sn,sg:it.sg,...macrosOf(f,g)};
      });
      return{slot,title:opt.n,items:scaled};
    });
    out.push({day:i+1,meals,total:tot(meals.flatMap(m=>m.items))});
  }
  return out;
}

function suggestNext(exId,defSets,defReps,defRpe){
  const last=lastFor(exId);
  if(!last)return{w:20,r:parseInt(defReps)||10,sets:defSets,note:'أول مرة — ابدأ بوزن مريح واضبط التكنيك'};
  const rpe=last.rpe||defRpe||8,maxR=parseInt(String(defReps).split('-').pop())||12;
  if(rpe<=7.5&&last.r>=maxR)return{w:r1(last.w+2.5),r:parseInt(defReps)||8,sets:defSets,note:`جبت ${last.r} ريب بـ RPE ${rpe} — زوّد 2.5 كجم`};
  if(rpe<=8.5&&last.r<maxR)return{w:last.w,r:Math.min(maxR,last.r+1),sets:defSets,note:`ثبّت ${last.w} كجم وحاول تجيب ${Math.min(maxR,last.r+1)} ريب`};
  if(rpe>=9.5)return{w:r1(Math.max(0,last.w-2.5)),r:last.r,sets:defSets,note:`آخر مرة كانت تقيلة (RPE ${rpe}) — خفف 2.5 كجم`};
  return{w:last.w,r:last.r,sets:defSets,note:`آخر مرة: ${last.w} كجم × ${last.r} (RPE ${rpe})`};
}
